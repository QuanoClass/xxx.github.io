      : makeDisabled('上一頁'));

    var visible = getPageNumbers(totalPages, pageNumber);
    visible.forEach(function (item) {
      if (item === '…') {
        addEllipsis(nav);
        return;
      }

      if (item === pageNumber) {
        var current = document.createElement('span');
        current.className = 'quano-pager-link quano-pager-number is-current';
        current.setAttribute('aria-current', 'page');
        current.textContent = String(item);
        nav.appendChild(current);
        return;
      }

      var href = urlMap[item];
      if (!href) {
        // If Blogger does not return a usable feed boundary, avoid creating
        // a misleading link. Native previous/next navigation remains available.
        nav.appendChild(makeDisabled(String(item)));
        return;
      }
      nav.appendChild(makeLink(String(item), href, 'quano-pager-number'));
    });

    nav.appendChild(pageNumber < totalPages && nativeLinks.next
      ? makeLink('下一頁', appendPageNumberToNativeUrl(nativeLinks.next, pageNumber + 1), 'quano-pager-next', '下一頁')
      : makeDisabled('下一頁'));
  }

  async function renderFromFeed() {
    var nativeLinks = getNativeNavigationLinks();
    var firstWindowPage = Math.max(2, currentPage - 2);
    var firstBoundaryIndex = (firstWindowPage - 1) * perPage;
    var slice = await fetchFeedSlice(firstBoundaryIndex, Math.min(35, perPage * 5));
    var totalPages = Math.max(1, Math.ceil(slice.total / perPage));
    var safeCurrentPage = Math.min(currentPage, totalPages);

    // Use one feed slice for the nearby numbers. If the last-page boundary is
    // outside that slice, retrieve only that single boundary entry.
    var entryByIndex = Object.create(null);
    slice.entries.forEach(function (entry, index) {
      entryByIndex[slice.startIndex + index] = entry;
    });

    var visible = getPageNumbers(totalPages, safeCurrentPage);
    var requiredPages = visible.filter(function (item) {
      return typeof item === 'number' && item > 1 && item !== safeCurrentPage;
    });
    var missingPages = requiredPages.filter(function (page) {
      return !entryByIndex[(page - 1) * perPage];
    });

    await Promise.all(missingPages.map(async function (page) {
      var boundaryIndex = (page - 1) * perPage;
      var one = await fetchFeedSlice(boundaryIndex, 1);
      if (one.entries.length) entryByIndex[boundaryIndex] = one.entries[0];
    }));

    var urlMap = Object.create(null);
    urlMap[1] = addPageNumber(basePageUrl.href, 1);

    requiredPages.forEach(function (page) {
      var entry = entryByIndex[(page - 1) * perPage];
      var publishedDate = getPublished(entry);
      var url = makeTargetUrl(page, publishedDate);
      if (url) urlMap[page] = url;
    });

    renderPager(totalPages, safeCurrentPage, urlMap, nativeLinks);
  }

  // Keep the current page indicator and native navigation usable even if
  // the public feed is temporarily unavailable.
  var staticCurrent = pager.querySelector('.is-current');
  if (staticCurrent) staticCurrent.textContent = String(currentPage);
  tagNativeLinksForFallback();

  renderFromFeed().catch(function (error) {
    // Deliberately fall back to Blogger's native pager without scanning pages.
    if (window.console && console.warn) console.warn('Quano pager fallback:', error);
  });
